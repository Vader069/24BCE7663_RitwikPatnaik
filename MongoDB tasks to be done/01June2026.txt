			Connection con=DriverManager.getConnection("jdbc:mysql://localhost:3306/vit_campus","root","Ritwik1902");
			System.out.println("Success");
			
			Statement s=con.createStatement();
			ResultSet r1=s.executeQuery("select * from cse");
			
			while(r1.next())
			{
				int sid=r1.getInt("Sid");
				String name=r1.getString("sname");
				int marks=r1.getInt("smarks");
				String dept=r1.getString("dept");
				String count=r1.getString("scountry");
				System.out.println(sid+" "+name+" "+marks+" "+dept+" "+count);
			}
		}
		catch(Exception e)
		{
			System.out.println(e.toString());
		}
	}

}

public void read_data(String sl_no){
	DB_connection obj_DB_Connection=new DB_connection();
	Connection connection=obj_DB_Connection.get_connection();
	PreparedStatement ps=null;
	ResultSet rs=null;
	try {
		String query="select * from user";
		ps=connection.prepareStatement(query);
		System.out.println(ps);
		rs=ps.executeQuery();
		
		System.out.println("SL_NO\t\tNAME\t\tEMAIL");
		System.out.println("--------------------------------------------------");
		
		while(rs.next()){
			System.out.println(rs.getString("sl_no") + "\t\t" + rs.getString("name") + "\t\t" + rs.getString("email"));
		}
	} catch (Exception e) {
		System.out.println(e);
	}
}


public void edit_data(String sl_no){
	DB_Connection obj_DB_Connection=new DB_Connection();
	Connection connection=obj_DB_Connection.get_connection();
	PreparedStatement ps=null;
	ResultSet rs=null;
	try{
		String query = "select * from user where sl_no=?";
		ps = connection.prepareStatement(query);
		ps.setString(1, sl_no);
		System.out.println(ps);
		rs=ps.executeQuery();
		if(rs.next()){
			System.out.println("Sl no -"+rs.getString("sl_no"));
			System.out.println("name -"+rs.getString("name"));
			System.out.println("email -"+rs.getString("email"));
			System.out.println("---------------");
			}
	}catch (Exception e) {
		System.out.println(e);
	}
}
